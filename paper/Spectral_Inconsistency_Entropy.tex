\begin{abstract}
Pairwise comparison data---arising in sports rankings, autonomous sensor fusion, peer review, and RLHF preference modeling---frequently exhibits cyclic inconsistencies. While classical combinatorial Hodge theory (HodgeRank) quantifies the total magnitude of inconsistency via the cyclic residual ratio $\rho(G, w) = \|r\|^2 / \|w\|^2$, it is invariant to how contradictions are topologically distributed across the comparison network. In this paper, we introduce \textbf{Spectral Inconsistency Entropy ($H$)}, a scale- and potential-invariant diagnostic defined as the Shannon entropy of the cyclic residual's energy projection across the distinct eigenspaces of the cycle-restricted 1-up-Laplacian $L_{\mathcal{Z}} = Z^T B_1 B_1^T Z$.

We establish four main results: (1) \textbf{Universal Clique Degeneracy (Theorem 1):} Using the simplicial Eckmann identity $B_0^T B_0 + B_1 B_1^T = n I_{|E|}$ on the clique 2-complex of $K_n$, we prove that $L_{\mathcal{Z}} = n I_{\beta_1(K_n)}$ possesses a single eigenvalue $\lambda = n$ of multiplicity $\beta_1(K_n) = (n-1)(n-2)/2$ for every $n \ge 3$, so $H(K_n, w) \equiv 0$ identically. (2) \textbf{Harmonic--Curl Separation and Scale Invariance (Theorem 2):} The $\lambda=0$ eigenspace of $L_{\mathcal{Z}}$ is canonically isomorphic to $\mathcal{H}^1(\mathcal{K}_2(G)) = \ker(B_0) \cap \ker(B_1^T)$, while $\lambda > 0$ eigenspaces span $\operatorname{im}(B_1)$, and $H(G, B_0^T s + \alpha \delta) = H(G, \delta)$ for all $s$ and $\alpha \neq 0$. (3) \textbf{Exact Isotropic Dirichlet--Digamma Asymptote (Theorem 3):} Under isotropic Gaussian cycle noise $c \sim \mathcal{N}(0, \sigma^2 I_{\beta_1})$, the eigenspace energies follow $\operatorname{Dirichlet}(d_1/2, \dots, d_{|\Lambda|}/2)$ with closed-form expected entropy $H_{\mathrm{iso}}(G) = \frac{1}{\ln 2} \sum_{\lambda \in \Lambda} \frac{d_\lambda}{\beta_1} [\psi(\beta_1/2 + 1) - \psi(d_\lambda/2 + 1)]$. (4) \textbf{Empirical Validation \& Cycle-Leverage Localization:} Across 1,000 Monte Carlo trials per edit distance $k$ on $G(15, 0.4)$, $H(G, w)$ increases strictly monotonically from $k=1$ ($3.7641 \pm 0.0394$ bits) to the exact Dirichlet--Digamma asymptote $H_{\mathrm{iso}} = 4.1175$ bits ($4.1274 \pm 0.0124$ bits at $k=40$), while cycle-leverage cosine alignment localizes single corrupted edges with $91.3\%$--$100.0\%$ Top-1 accuracy across five canonical graph families.
\end{abstract}

\section{Introduction}
Given a comparison graph $G=(V,E)$ with $|V|=n$ and $|E|=m$, an antisymmetric edge flow $w \in \mathbb{R}^m$ is consistent if and only if $w = B_0^T s$ for some node potential $s \in \mathbb{R}^n$. Combinatorial Hodge theory \cite{jiang2011} decomposes $\mathbb{R}^m$ orthogonally as
\[
w = B_0^T s^* \oplus r_{\mathrm{curl}} \oplus r_{\mathrm{harm}}, \qquad r = r_{\mathrm{curl}} \oplus r_{\mathrm{harm}} = \Pi_{\mathcal{Z}} w \in \mathcal{Z} = \ker(B_0).
\]
While the classical inconsistency ratio $\rho(G,w) = \|r\|_2^2 / \|w\|_2^2$ measures total inconsistency magnitude, it confounds the ground-truth potential scale $\|B_0^T s^*\|_2$ with the fault topology and cannot distinguish a single corrupted sensor ($L_0$ edit distance $k=1$) from systemic network-wide noise ($k \sim m$).

\section{Mathematical Framework and Main Theorems}
Let $\mathcal{K}_2(G) = (V, E, T)$ be the clique 2-complex of $G$ with boundary matrices $B_0 \in \mathbb{R}^{|V|\times |E|}$ and $B_1 \in \mathbb{R}^{|E|\times |T|}$. Let $Z \in \mathbb{R}^{|E| \times \beta_1(G)}$ be an orthonormal basis of $\mathcal{Z} = \ker(B_0)$, where $\beta_1(G) = |E| - |V| + c(G)$. The cycle-restricted 1-up-Laplacian is $L_{\mathcal{Z}} = Z^T B_1 B_1^T Z$. Let $\Lambda(G)$ be its distinct eigenvalues with multiplicities $d_\lambda$ and orthogonal projectors $P_\lambda$.

\begin{definition}[Spectral Inconsistency Entropy]
For $w \in \mathbb{R}^{|E|}$ with $c = Z^T w \neq 0$, let $p_\lambda(G,w) = \|P_\lambda c\|_2^2 / \|c\|_2^2$. The Spectral Inconsistency Entropy is
\[
H(G, w) = -\sum_{\lambda \in \Lambda(G)} p_\lambda(G,w) \log_2 p_\lambda(G,w) \in [0, \log_2 |\Lambda(G)|].
\]
\end{definition}

\begin{theorem}[Universal Clique Degeneracy via Eckmann Identity]
For every complete graph $K_n$ ($n \ge 3$), the 1-Hodge Laplacian satisfies $B_0^T B_0 + B_1 B_1^T = n I_{|E|}$. Consequently, $L_{\mathcal{Z}} = n I_{\beta_1(K_n)}$ with $\beta_1(K_n) = (n-1)(n-2)/2$ and single eigenvalue $\lambda = n$. Thus $H(K_n, w) \equiv 0$ identically for all $n \ge 3$ and all $w$.
\end{theorem}

\begin{theorem}[Harmonic--Curl Separation \& Scale/Potential Invariance]
(1) The $\lambda=0$ eigenspace of $L_{\mathcal{Z}}$ maps via $Z$ isomorphically onto $\mathcal{H}^1(\mathcal{K}_2(G)) = \ker(B_0) \cap \ker(B_1^T)$, while $\bigoplus_{\lambda > 0} V_\lambda$ maps onto $\operatorname{im}(B_1)$. (2) For all $s \in \mathbb{R}^{|V|}$, $\delta \notin \operatorname{im}(B_0^T)$, and $\alpha \neq 0$, $H(G, B_0^T s + \alpha \delta) = H(G, \delta)$.
\end{theorem}

\begin{theorem}[Exact Isotropic Dirichlet--Digamma Asymptote]
Under isotropic Gaussian cycle noise $c \sim \mathcal{N}(0, \sigma^2 I_{\beta_1(G)})$, the energy fractions follow $(p_\lambda)_{\lambda \in \Lambda} \sim \operatorname{Dirichlet}(d_1/2, \dots, d_{|\Lambda|}/2)$ with exact expected entropy
\[
H_{\mathrm{iso}}(G) = \frac{1}{\ln 2} \sum_{\lambda \in \Lambda(G)} \frac{d_\lambda}{\beta_1(G)} \left[ \psi\!\left(\frac{\beta_1(G)}{2} + 1\right) - \psi\!\left(\frac{d_\lambda}{2} + 1\right) \right] \le H_{\mathrm{sys}}(G) = -\sum_{\lambda \in \Lambda(G)} \frac{d_\lambda}{\beta_1(G)} \log_2 \frac{d_\lambda}{\beta_1(G)}.
\]
\end{theorem}
